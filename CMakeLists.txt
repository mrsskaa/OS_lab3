cmake_minimum_required(VERSION 3.16)
project(OS_lab3 C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(os_lab3 src/main.c src/minmax.c)
target_compile_definitions(os_lab3 PRIVATE _GNU_SOURCE)
target_link_libraries(os_lab3 PRIVATE Threads::Threads)
target_compile_options(os_lab3 PRIVATE -Wall -Wextra -Wpedantic)
